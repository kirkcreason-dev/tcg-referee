function finishSetup(){

  if(!selGame)return;

  localStorage.setItem("tcgGame",selGame);

  document.getElementById("setupScreen").style.display="none";

  document.getElementById("mainApp").style.display="flex";

  applyGame(selGame);

}

function changeGame(){

  if(confirm("Change game? Your current LP and match data will be reset.")){

    localStorage.removeItem("tcgGame");

  }

}

function applyGame(g){

}

function scg(g,el){

  cGame=g;

  document.querySelectorAll("#pan-cards .pill").forEach(function(p){p.classList.remove("on");});

  el.classList.add("on");

function scard(){

  var q=document.getElementById("cs").value.trim();if(!q)return;

  var key=cGame+":"+q.toLowerCase();

  if(cCache[key]){showCard(cCache[key]);return;}

  document.getElementById("cr").innerHTML="<div style='color:#9090a8;font-size:12px;padding:16px;text-align:center'>Searching...</div>";

  if(cGame==="ygo")fygo(q,key);

  else if(cGame==="pkm")fpkm(q,key);

  else if(cGame==="mtg")fmtg(q,key);

}

function fygo(q,k){

  fetch("https://db.ygoprodeck.com/api/v7/cardinfo.php?fname="+encodeURIComponent(q))

  .then(function(r){return r.json();})

  .then(function(d){

    if(!d.data||!d.data.length){document.getElementById("cr").innerHTML="<div style='color:#505068;font-size:12px;padding:16px;text-align:center'>Card not found. Check spelling.</div>";return;}

    var c=d.data[0];

    var tags=[c.type||""];

    if(c.attribute)tags.push(c.attribute);if(c.race)tags.push(c.race);

    if(c.level)tags.push("Level "+c.level);if(c.atk!==undefined)tags.push("ATK: "+c.atk);if(c.def!==undefined)tags.push("DEF: "+c.def);

    if(c.linkval)tags.push("Link-"+c.linkval);if(c.rank)tags.push("Rank "+c.rank);

    var card={name:c.name,tags:tags,text:c.desc||"No text."};

    cCache[k]=card;showCard(card);

  }).catch(function(){document.getElementById("cr").innerHTML="<div style='color:#e05050;font-size:12px;padding:16px;text-align:center'>Search failed. Check internet.</div>";});

}

function fpkm(q,k){

  fetch("https://api.pokemontcg.io/v2/cards?q=name:"+encodeURIComponent(q)+"&pageSize=1")

  .then(function(r){return r.json();})

  .then(function(d){

    if(!d.data||!d.data.length){document.getElementById("cr").innerHTML="<div style='color:#505068;font-size:12px;padding:16px;text-align:center'>Card not found.</div>";return;}

    var c=d.data[0];

    var tags=[c.supertype||""];

    if(c.hp)tags.push("HP: "+c.hp);if(c.types)tags.push(c.types.join("/"));

    if(c.set&&c.set.name)tags.push(c.set.name);

    var text="";

    if(c.abilities)c.abilities.forEach(function(a){text+="[Ability] "+a.name+": "+a.text+"\n\n";});

    if(c.attacks)c.attacks.forEach(function(a){text+=a.name+(a.damage?" - "+a.damage:"")+"\n"+(a.text||"")+"\n\n";});

    var card={name:c.name,tags:tags,text:text.trim()||"No text."};

    cCache[k]=card;showCard(card);

  }).catch(function(){document.getElementById("cr").innerHTML="<div style='color:#e05050;font-size:12px;padding:16px;text-align:center'>Search failed.</div>";});

}

function fmtg(q,k){

  fetch("https://api.scryfall.com/cards/named?fuzzy="+encodeURIComponent(q))

  .then(function(r){return r.json();})

  .then(function(c){

    if(c.object==="error"){document.getElementById("cr").innerHTML="<div style='color:#505068;font-size:12px;padding:16px;text-align:center'>Card not found.</div>";return;}

    var tags=[];

    if(c.type_line)tags.push(c.type_line);if(c.mana_cost)tags.push(c.mana_cost);

    if(c.power)tags.push("P/T: "+c.power+"/"+c.toughness);if(c.loyalty)tags.push("Loyalty: "+c.loyalty);

    var text=c.oracle_text||"";

    if(c.card_faces)text=c.card_faces.map(function(f){return f.name+"\n"+f.oracle_text;}).join("\n---\n");

    var card={name:c.name,tags:tags,text:text||"No text."};

    cCache[k]=card;showCard(card);

  }).catch(function(){document.getElementById("cr").innerHTML="<div style='color:#e05050;font-size:12px;padding:16px;text-align:center'>Search failed.</div>";});

}

function showCard(card){

  var th=card.tags.filter(function(t){return t;}).map(function(t){return "<span class='ctag'>"+t+"</span>";}).join("");

  var tx=card.text.replace(/\n/g,"<br>");

  document.getElementById("cr").innerHTML="<div class='card' style='border-color:rgba(232,216,138,0.2)'><div class='cn'>"+card.name+"</div><div style='display:flex;flex-wrap:wrap;margin-bottom:4px'>"+th+"</div><div class='cx'>"+tx+"</div></div>";

}

function sj(){

  var inp=document.getElementById("ji");

  var q=inp.value.trim();if(!q)return;

  inp.value="";inp.style.height="36px";

  jmsg("user",q,"");

  document.getElementById("isb").disabled=true;


  // STEP 1: Check card-specific rulings database first

  var cardRuling=findCardRuling(q,jGame);

  if(cardRuling){

    jmsg("ai",cardRuling.a,"Official Ruling");

    document.getElementById("isb").disabled=false;

    return;

  }


  // STEP 2: Build context from rules database

  var stop={a:1,i:1,my:1,by:1,to:1,if:1,of:1,it:1,in:1,is:1,do:1,the:1,and:1,what:1,when:1,does:1,can:1,how:1,that:1,this:1,with:1,from:1,your:1,have:1,will:1,its:1,card:1,cards:1,player:1,opponent:1};

  var wds=q.toLowerCase().replace(/[^a-z0-9 ]/g," ").split(" ").filter(function(w){return w.length>2&&!stop[w];});

  var exp={};

  wds.forEach(function(w){

    exp[w]=true;

    if(SYNS[w])SYNS[w].split(" ").forEach(function(s){exp[s]=true;});

    Object.keys(SYNS).forEach(function(k){

      if(k.indexOf(w)>=0||w.indexOf(k)>=0){exp[k]=true;SYNS[k].split(" ").forEach(function(s){exp[s]=true;});}

    });

  });

  var games=jGame==="all"?Object.keys(RULES):[jGame];

  var res=[];

  games.forEach(function(g){

    (RULES[g]||[]).forEach(function(r){

      var ti=r.t.toLowerCase(),su=r.s.toLowerCase(),de=r.e.toLowerCase();

      var sc=0;

      wds.forEach(function(w){if(ti.indexOf(w)>=0)sc+=15;if(su.indexOf(w)>=0)sc+=8;if(de.indexOf(w)>=0)sc+=3;});

      Object.keys(exp).forEach(function(w){if(w.length<3)return;if(ti.indexOf(w)>=0)sc+=5;if(su.indexOf(w)>=0)sc+=3;if(de.indexOf(w)>=0)sc+=1;});

      for(var i=0;i<wds.length-1;i++){var ph=wds[i]+" "+wds[i+1];if(ti.indexOf(ph)>=0)sc+=25;if(su.indexOf(ph)>=0)sc+=12;if(de.indexOf(ph)>=0)sc+=6;}

      if(sc>5)res.push({r:r,s:sc});

    });

  });

  res.sort(function(a,b){return b.s-a.s;});

  var context=res.slice(0,8).map(function(x){return "RULE - "+x.r.t+":\n"+x.r.e;}).join("\n\n");


  // Show thinking indicator

  var msgs=document.getElementById("jm");

  var tp=document.createElement("div");tp.className="msg ma";tp.id="tp";

  tp.innerHTML="<div class='bub' style='color:#505068'>Thinking...</div>";

  msgs.appendChild(tp);msgs.scrollTop=msgs.scrollHeight;


    var gl={all:"Yu-Gi-Oh!, Pokemon TCG, Magic The Gathering, Digimon TCG, Gundam Card Game, and One Piece Card Game",ygo:"Yu-Gi-Oh!",pkm:"Pokemon TCG",mtg:"Magic The Gathering",dig:"Digimon TCG",gundam:"Gundam Card Game",op:"One Piece Card Game"};

  var sys="You are a certified Head Judge for "+gl[jGame]+" tournaments with 10+ years experience. You have complete memorized knowledge of every official ruling, errata, and interaction.\n\nCRITICAL FACTS YOU MUST KNOW:\n- Costs are paid at activation before the chain resolves. Cards that leave the field as a cost are gone before effects resolve.\n- Skill Drain only negates face-up monsters ON THE FIELD. If a monster left the field as cost it cannot be negated.\n- Sangan only searches if sent to the GY. If banished instead it does not search.\n- Super Polymerization is a Quick-Play Spell Speed 2. It CAN be chained to other effects.\n- Normal Spells resolve even if destroyed after activation. Continuous Spells need to stay on field.\n- Missing the Timing: optional WHEN triggers miss timing if their trigger was not the last thing to happen.\n- IF triggers never miss timing.\n- Chain resolves Last In First Out.\n- D.D. Crow banishes from GY. If it banishes a monster that was a cost, that monster is now banished not in GY.\n\nRULING PROCESS:\n1. Identify all cards mentioned\n2. Check costs vs effects (semicolon rule)\n3. Check field presence requirements\n4. Check GY vs banished destination\n5. Give definitive yes or no ruling with exact reason\n\nKeep answer under 150 words. Plain text only. Be definitive - never say maybe.";

  if(context)sys+="\n\nOFFICIAL RULES FROM RULEBOOK:\n"+context;


  fetch("https://tcg-judge.kirkcreason.workers.dev",{

    method:"POST",headers:{"Content-Type":"application/json"},

    body:JSON.stringify({system:sys,question:q})

  }).then(function(r){return r.json();})

  .then(function(d){

    var t=document.getElementById("tp");if(t)t.remove();

    var ans=(d.answer||d.error||"No response.").replace(/\n\n/g,"<br><br>").replace(/\n/g,"<br>");

    jmsg("ai",ans,"AI Judge");

  }).catch(function(){

    var t=document.getElementById("tp");if(t)t.remove();

  }).finally(function(){document.getElementById("isb").disabled=false;});

}
